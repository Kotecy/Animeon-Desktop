// ==UserScript==
// @name         AnimeOn Anomaly Watcher v1.2
// @namespace    animeon-research
// @version      1.2
// @description  Наблюдение за аномалией: подсветка + оригинальный звук из Kaspersky
// @match        https://animeon.cc/*
// @match        https://www.animeon.cc/*
// @run-at       document-start
// @resource     KasperskyPig https://1997.kaspersky.ru/static/audio/infected.wav
// @grant        GM_getResourceURL
// ==/UserScript==

(() => {
    "use strict";

    const VERSION = "1.2";
    const SETTINGS_KEY = "animeon_anomaly_watcher_v1_2_settings";

    const STATE_URL = "/api/event/boar/anomaly/state";
    const ANOMALY_SELECTOR =
        'button.anomaly-orb-root,button[aria-label="Аномалия — собрать награду"]';

    // Адаптивный read-only polling состояния события.
    const STATE_POLL_READY_MS = 4_000;
    const STATE_POLL_WAIT_ELIGIBLE_MS = 12_000;
    const STATE_POLL_EVENT_INACTIVE_MS = 60_000;
    const STATE_POLL_NO_REWARDS_MS = 60_000;
    const STATE_POLL_NO_STATE_MS = 5_000;
    const STATE_POLL_ERROR_MS = 15_000;
    const STATE_POLL_COOLDOWN_MARGIN_MS = 30_000;
    const STATE_POLL_AFTER_COOLDOWN_MS = 750;

    // На скрытой вкладке даём штатному коду AnimeOn короткое окно visible,
    // чтобы его собственный visibilitychange/refetch мог обновить anomaly-store.
    const SITE_VISIBILITY_BRIDGE_MS = 7_000;

    // Подтверждение исчезновения защищает от повторного звука при React rerender.
    const ANOMALY_DISAPPEAR_CONFIRM_MS = 4_000;

    // Фоновый helper из старшей ветки скрипта.
    const KEEPALIVE_LOCK_NAME = "animeon-anomaly-watcher-v12";
    const KEEPALIVE_RTC_PING_MS = 20_000;

    const PANEL_WATCH_MS = 750;
    const PANEL_REFRESH_MS = 1_000;

    const rawFetch = window.fetch.bind(window);

    // ------------------------------------------------------------
    // Settings
    // ------------------------------------------------------------

    function defaultSettings() {
        return {
            watchEnabled: true,
            soundEnabled: true,
            panel: {
                x: null,
                y: null,
                collapsed: false
            }
        };
    }

    function loadSettings() {
        const defaults = defaultSettings();

        try {
            const parsed = JSON.parse(
                localStorage.getItem(SETTINGS_KEY) || "null"
            );

            if (!parsed) return defaults;

            return {
                ...defaults,
                ...parsed,
                panel: {
                    ...defaults.panel,
                    ...(parsed.panel || {})
                }
            };
        } catch {
            return defaults;
        }
    }

    let settings = loadSettings();

    function saveSettings() {
        try {
            localStorage.setItem(
                SETTINGS_KEY,
                JSON.stringify(settings)
            );
        } catch {}

        updatePanel();
    }

    // ------------------------------------------------------------
    // Runtime
    // ------------------------------------------------------------

    const runtime = {
        serverState: null,
        serverStateAt: 0,
        cooldownUntil: 0,
        stateFetchInFlight: false,
        statePollTimer: null,
        nextStateCheckAt: 0,
        statePollMode: "init",
        lastCheckStatus: "—",

        anomalyPresent: false,
        anomalyMissingSince: 0,
        currentButton: null,
        alertedCurrentEligibility: false,
        lastAlertAt: 0,

        manualStatus: "",

        audio: null,
        audioPrimed: false,
        audioState: "waiting",

        siteVisibilityBridge: {
            active: false,
            timer: null,
            startedAt: 0
        },

        keepAlive: {
            enabled: false,
            rtcState: "off",
            lockState: "off",
            pc1: null,
            pc2: null,
            dc1: null,
            dc2: null,
            rtcPingTimer: null,
            lockRelease: null
        }
    };

    function clamp(value, min, max) {
        return Math.min(max, Math.max(min, value));
    }

    function formatClock(date = new Date()) {
        return date.toLocaleTimeString([], {
            hour: "2-digit",
            minute: "2-digit",
            second: "2-digit"
        });
    }

    // ------------------------------------------------------------
    // Real visibility + temporary bridge for the site
    // ------------------------------------------------------------

    const nativeHiddenGetter =
        Object.getOwnPropertyDescriptor(
            Document.prototype,
            "hidden"
        )?.get || null;

    const nativeVisibilityStateGetter =
        Object.getOwnPropertyDescriptor(
            Document.prototype,
            "visibilityState"
        )?.get || null;

    function isActuallyHidden() {
        try {
            if (nativeHiddenGetter) {
                return !!nativeHiddenGetter.call(document);
            }
        } catch {}

        try {
            if (nativeVisibilityStateGetter) {
                return (
                    nativeVisibilityStateGetter.call(document) === "hidden"
                );
            }
        } catch {}

        return document.visibilityState === "hidden";
    }

    function restoreSiteVisibility() {
        const bridge = runtime.siteVisibilityBridge;

        if (bridge.timer) {
            clearTimeout(bridge.timer);
            bridge.timer = null;
        }

        if (!bridge.active) return;

        try {
            delete document.hidden;
        } catch {}

        try {
            delete document.visibilityState;
        } catch {}

        bridge.active = false;
        bridge.startedAt = 0;
        updatePanel();
    }

    function activateSiteVisibilityBridge() {
        if (!settings.watchEnabled) return false;
        if (!isActuallyHidden()) return false;

        const bridge = runtime.siteVisibilityBridge;

        if (bridge.timer) {
            clearTimeout(bridge.timer);
            bridge.timer = null;
        }

        let changed = false;

        try {
            Object.defineProperty(document, "hidden", {
                configurable: true,
                enumerable: true,
                get: () => false
            });

            changed = changed || document.hidden === false;
        } catch {}

        try {
            Object.defineProperty(document, "visibilityState", {
                configurable: true,
                enumerable: true,
                get: () => "visible"
            });

            changed = changed || document.visibilityState === "visible";
        } catch {}

        if (!changed) return false;

        bridge.active = true;
        bridge.startedAt = Date.now();

        try {
            document.dispatchEvent(new Event("visibilitychange"));
        } catch {}

        bridge.timer = setTimeout(
            restoreSiteVisibility,
            SITE_VISIBILITY_BRIDGE_MS
        );

        updatePanel();
        return true;
    }

    // ------------------------------------------------------------
    // State polling
    // ------------------------------------------------------------

    function applyServerState(body) {
        if (!body || typeof body !== "object") return;

        const previousEligible =
            runtime.serverState?.eligible === true;

        runtime.serverState = body;
        runtime.serverStateAt = Date.now();

        const cooldownMs = Math.max(
            0,
            Number(body.cooldown_ms) || 0
        );

        runtime.cooldownUntil = Date.now() + cooldownMs;

        const eligibleNow = body.eligible === true;

        // После ручного сбора сайт обычно переводит eligible в false.
        // Это вооружает звук для следующей реальной аномалии.
        if (previousEligible && !eligibleNow) {
            runtime.alertedCurrentEligibility = false;
        }

        if (
            settings.watchEnabled &&
            eligibleNow &&
            isActuallyHidden()
        ) {
            activateSiteVisibilityBridge();
        }

        if (
            !eligibleNow &&
            runtime.siteVisibilityBridge.active
        ) {
            restoreSiteVisibility();
        }

        scheduleAdaptiveStateCheck();
        updatePanel();
    }

    async function fetchState(reason = "timer") {
        if (runtime.stateFetchInFlight) {
            return runtime.serverState;
        }

        runtime.stateFetchInFlight = true;
        runtime.lastCheckStatus = "проверка...";
        updatePanel();

        try {
            const response = await rawFetch(STATE_URL, {
                method: "GET",
                credentials: "include",
                cache: "no-store"
            });

            const body = await response.json();

            if (response.ok && body && typeof body === "object") {
                applyServerState(body);
                runtime.lastCheckStatus =
                    `${response.status} · ${formatClock()}`;
            } else {
                runtime.lastCheckStatus =
                    `${response.status} · ошибка`;
            }

            if (reason === "manual") {
                runtime.manualStatus = response.ok
                    ? `Состояние обновлено · ${formatClock()}`
                    : `State вернул HTTP ${response.status}`;
            }

            return body;
        } catch {
            runtime.lastCheckStatus = `ошибка · ${formatClock()}`;

            if (reason === "manual") {
                runtime.manualStatus =
                    "Не удалось получить состояние";
            }

            if (settings.watchEnabled) {
                scheduleAdaptiveStateCheck(
                    STATE_POLL_ERROR_MS,
                    "error-retry"
                );
            }

            return null;
        } finally {
            runtime.stateFetchInFlight = false;
            updatePanel();
        }
    }

    function computeAdaptiveStatePollPlan() {
        const state = runtime.serverState;
        const now = Date.now();

        if (!settings.watchEnabled) {
            return {
                delayMs: null,
                mode: "off"
            };
        }

        if (!state) {
            return {
                delayMs: STATE_POLL_NO_STATE_MS,
                mode: "no-state"
            };
        }

        if (state.event_active !== true) {
            return {
                delayMs: STATE_POLL_EVENT_INACTIVE_MS,
                mode: "event-inactive"
            };
        }

        if (
            state.remaining_today != null &&
            Number(state.remaining_today) <= 0
        ) {
            return {
                delayMs: STATE_POLL_NO_REWARDS_MS,
                mode: "no-rewards"
            };
        }

        const cooldownLeft = Math.max(
            0,
            runtime.cooldownUntil - now
        );

        if (cooldownLeft > 60_000) {
            return {
                delayMs: Math.max(
                    15_000,
                    cooldownLeft - STATE_POLL_COOLDOWN_MARGIN_MS
                ),
                mode: "long-cooldown"
            };
        }

        if (cooldownLeft > 0) {
            return {
                delayMs: Math.max(
                    2_500,
                    cooldownLeft + STATE_POLL_AFTER_COOLDOWN_MS
                ),
                mode: "cooldown-ending"
            };
        }

        const eligible = state.eligible === true;
        const chance = Number(state.spawn_chance) || 0;

        if (eligible && chance > 0) {
            return {
                delayMs: STATE_POLL_READY_MS,
                mode: "ready"
            };
        }

        if (!eligible) {
            return {
                delayMs: STATE_POLL_WAIT_ELIGIBLE_MS,
                mode: "waiting-eligible"
            };
        }

        return {
            delayMs: 15_000,
            mode: "fallback"
        };
    }

    function scheduleAdaptiveStateCheck(
        forcedDelayMs = null,
        forcedMode = null
    ) {
        if (runtime.statePollTimer) {
            clearTimeout(runtime.statePollTimer);
            runtime.statePollTimer = null;
        }

        if (!settings.watchEnabled) {
            runtime.nextStateCheckAt = 0;
            runtime.statePollMode = "off";
            updatePanel();
            return;
        }

        const plan =
            forcedDelayMs != null
                ? {
                    delayMs: Math.max(
                        500,
                        Number(forcedDelayMs) || 500
                    ),
                    mode: forcedMode || "forced"
                }
                : computeAdaptiveStatePollPlan();

        if (plan.delayMs == null) {
            runtime.nextStateCheckAt = 0;
            runtime.statePollMode = plan.mode;
            updatePanel();
            return;
        }

        const delayMs = Math.max(
            500,
            Math.round(plan.delayMs)
        );

        runtime.nextStateCheckAt = Date.now() + delayMs;
        runtime.statePollMode = plan.mode;

        runtime.statePollTimer = setTimeout(() => {
            runtime.statePollTimer = null;
            runtime.nextStateCheckAt = 0;

            if (!settings.watchEnabled) return;

            fetchState("timer").finally(() => {
                if (
                    settings.watchEnabled &&
                    !runtime.statePollTimer
                ) {
                    scheduleAdaptiveStateCheck();
                }
            });
        }, delayMs);

        updatePanel();
    }

    // ------------------------------------------------------------
    // Original Kaspersky infected.wav
    // ------------------------------------------------------------

    function getPigAudio() {
        if (runtime.audio) return runtime.audio;

        try {
            const resourceUrl = GM_getResourceURL("KasperskyPig");
            const audio = new Audio(resourceUrl);
            audio.preload = "auto";
            audio.volume = 1;

            runtime.audio = audio;
            return audio;
        } catch {
            runtime.audioState = "resource-error";
            return null;
        }
    }

    async function primeAudio() {
        if (!settings.soundEnabled) return false;
        if (runtime.audioPrimed) return true;

        const audio = getPigAudio();
        if (!audio) {
            updatePanel();
            return false;
        }

        const oldVolume = audio.volume;

        try {
            audio.pause();
            audio.currentTime = 0;
            audio.volume = 0;

            await audio.play();
            audio.pause();
            audio.currentTime = 0;
            audio.volume = oldVolume;

            runtime.audioPrimed = true;
            runtime.audioState = "ready";
            updatePanel();
            return true;
        } catch {
            audio.volume = oldVolume;
            runtime.audioState = "blocked";
            updatePanel();
            return false;
        }
    }

    async function playPigSound() {
        if (!settings.soundEnabled) return;

        const audio = getPigAudio();
        if (!audio) {
            runtime.manualStatus =
                "Не удалось загрузить infected.wav";
            updatePanel();
            return;
        }

        try {
            audio.pause();
            audio.currentTime = 0;
            audio.volume = 1;

            await audio.play();

            runtime.audioPrimed = true;
            runtime.audioState = "ready";
        } catch {
            runtime.audioState = "blocked";
            runtime.manualStatus =
                "Браузер заблокировал звук. Один раз кликните по странице.";
        }

        updatePanel();
    }

    function installAudioUnlock() {
        const unlock = () => {
            primeAudio();
        };

        document.addEventListener(
            "pointerdown",
            unlock,
            { once: true, capture: true }
        );

        document.addEventListener(
            "keydown",
            unlock,
            { once: true, capture: true }
        );
    }

    installAudioUnlock();

    // ------------------------------------------------------------
    // Anomaly DOM watcher + highlight
    // ------------------------------------------------------------

    function clearHighlightFromOldButtons(current = null) {
        const highlighted = document.querySelectorAll(
            ".aon-watcher-v12-highlight"
        );

        for (const element of highlighted) {
            if (element !== current) {
                element.classList.remove(
                    "aon-watcher-v12-highlight"
                );
            }
        }
    }

    function alertForAnomaly() {
        if (!settings.watchEnabled) return;
        if (runtime.alertedCurrentEligibility) return;

        runtime.alertedCurrentEligibility = true;
        runtime.lastAlertAt = Date.now();
        runtime.manualStatus =
            `АНОМАЛИЯ ОБНАРУЖЕНА · ${formatClock()}`;

        if (settings.soundEnabled) {
            playPigSound();
        }

        updatePanel();
    }

    function handleAnomalyButton(button) {
        if (!button || !button.isConnected) return;

        runtime.currentButton = button;
        runtime.anomalyMissingSince = 0;

        clearHighlightFromOldButtons(button);

        if (settings.watchEnabled) {
            button.classList.add(
                "aon-watcher-v12-highlight"
            );
        }

        if (!runtime.anomalyPresent) {
            runtime.anomalyPresent = true;
            alertForAnomaly();
        }

        updatePanel();
    }

    function confirmAnomalyMissing() {
        if (!runtime.anomalyPresent) return;

        if (!runtime.anomalyMissingSince) {
            runtime.anomalyMissingSince = Date.now();
            return;
        }

        if (
            Date.now() - runtime.anomalyMissingSince <
            ANOMALY_DISAPPEAR_CONFIRM_MS
        ) {
            return;
        }

        runtime.anomalyPresent = false;
        runtime.currentButton = null;
        runtime.anomalyMissingSince = 0;

        // Если server state уже подтверждает eligible=false, applyServerState
        // также сбросит флаг. При отсутствии state исчезновение DOM считается
        // достаточным признаком для вооружения следующего обнаружения.
        if (runtime.serverState?.eligible !== true) {
            runtime.alertedCurrentEligibility = false;
        }

        clearHighlightFromOldButtons();
        updatePanel();
    }

    function scanAnomalyDom() {
        if (!settings.watchEnabled) return;

        const button = document.querySelector(
            ANOMALY_SELECTOR
        );

        if (button) {
            handleAnomalyButton(button);
            return;
        }

        confirmAnomalyMissing();
    }

    function installAnomalyObserver() {
        if (!document.documentElement) {
            setTimeout(installAnomalyObserver, 50);
            return;
        }

        const observer = new MutationObserver(
            scanAnomalyDom
        );

        observer.observe(document.documentElement, {
            childList: true,
            subtree: true
        });

        scanAnomalyDom();
    }

    installAnomalyObserver();
    setInterval(scanAnomalyDom, 500);

    // ------------------------------------------------------------
    // Background keep-alive helper
    // ------------------------------------------------------------

    function keepAliveSummary() {
        const k = runtime.keepAlive;

        return (
            `${k.rtcState}/${k.lockState}` +
            (isActuallyHidden() ? " · hidden" : " · visible")
        );
    }

    function cleanupRtcKeepAlive() {
        const k = runtime.keepAlive;

        if (k.rtcPingTimer) {
            clearInterval(k.rtcPingTimer);
            k.rtcPingTimer = null;
        }

        for (const dc of [k.dc1, k.dc2]) {
            try {
                dc?.close();
            } catch {}
        }

        for (const pc of [k.pc1, k.pc2]) {
            try {
                pc?.close();
            } catch {}
        }

        k.dc1 = null;
        k.dc2 = null;
        k.pc1 = null;
        k.pc2 = null;
        k.rtcState = "off";
        updatePanel();
    }

    async function startRtcKeepAlive() {
        const k = runtime.keepAlive;

        if (
            !k.enabled ||
            k.rtcState === "starting" ||
            k.rtcState === "open"
        ) {
            return;
        }

        if (typeof RTCPeerConnection !== "function") {
            k.rtcState = "unsupported";
            updatePanel();
            return;
        }

        cleanupRtcKeepAlive();
        k.rtcState = "starting";
        updatePanel();

        try {
            const pc1 = new RTCPeerConnection({
                iceServers: []
            });

            const pc2 = new RTCPeerConnection({
                iceServers: []
            });

            k.pc1 = pc1;
            k.pc2 = pc2;

            pc1.onicecandidate = event => {
                if (event.candidate) {
                    pc2.addIceCandidate(
                        event.candidate
                    ).catch(() => {});
                }
            };

            pc2.onicecandidate = event => {
                if (event.candidate) {
                    pc1.addIceCandidate(
                        event.candidate
                    ).catch(() => {});
                }
            };

            pc2.ondatachannel = event => {
                k.dc2 = event.channel;
                k.dc2.onmessage = () => {};
            };

            const dc = pc1.createDataChannel(
                "animeon-anomaly-watcher-keepalive",
                { ordered: false }
            );

            k.dc1 = dc;

            dc.onopen = () => {
                if (!k.enabled) {
                    cleanupRtcKeepAlive();
                    return;
                }

                k.rtcState = "open";

                if (k.rtcPingTimer) {
                    clearInterval(k.rtcPingTimer);
                }

                k.rtcPingTimer = setInterval(() => {
                    try {
                        if (dc.readyState === "open") {
                            dc.send(String(Date.now()));
                        }
                    } catch {}
                }, KEEPALIVE_RTC_PING_MS);

                updatePanel();
            };

            dc.onclose = () => {
                if (!k.enabled) return;

                k.rtcState = "closed";
                updatePanel();

                setTimeout(() => {
                    if (k.enabled) {
                        startRtcKeepAlive();
                    }
                }, 1500);
            };

            dc.onerror = () => {
                k.rtcState = "error";
                updatePanel();
            };

            const offer = await pc1.createOffer();
            await pc1.setLocalDescription(offer);
            await pc2.setRemoteDescription(offer);

            const answer = await pc2.createAnswer();
            await pc2.setLocalDescription(answer);
            await pc1.setRemoteDescription(answer);
        } catch {
            k.rtcState = "error";
            cleanupRtcKeepAlive();
        }

        updatePanel();
    }

    function startWebLockKeepAlive() {
        const k = runtime.keepAlive;

        if (
            !k.enabled ||
            k.lockState === "waiting" ||
            k.lockState === "held"
        ) {
            return;
        }

        if (!navigator.locks?.request) {
            k.lockState = "unsupported";
            updatePanel();
            return;
        }

        k.lockState = "waiting";

        let release;

        const hold = new Promise(resolve => {
            release = resolve;
        });

        k.lockRelease = release;

        navigator.locks
            .request(
                KEEPALIVE_LOCK_NAME,
                { mode: "exclusive" },
                async () => {
                    if (!k.enabled) return;

                    k.lockState = "held";
                    updatePanel();

                    await hold;

                    k.lockState = "released";
                    updatePanel();
                }
            )
            .catch(() => {
                k.lockState = "error";
                updatePanel();
            });
    }

    function startBackgroundKeepAlive() {
        const k = runtime.keepAlive;

        if (k.enabled) return;

        k.enabled = true;
        startWebLockKeepAlive();
        startRtcKeepAlive();
        updatePanel();
    }

    function stopBackgroundKeepAlive() {
        const k = runtime.keepAlive;

        if (!k.enabled) return;

        k.enabled = false;

        if (typeof k.lockRelease === "function") {
            try {
                k.lockRelease();
            } catch {}
        }

        k.lockRelease = null;
        k.lockState = "off";

        cleanupRtcKeepAlive();
        updatePanel();
    }

    // ------------------------------------------------------------
    // Start / stop watcher
    // ------------------------------------------------------------

    function startWatcher() {
        settings.watchEnabled = true;
        runtime.manualStatus = "Наблюдение запущено";

        saveSettings();
        startBackgroundKeepAlive();
        primeAudio();

        fetchState("start").finally(() => {
            if (
                settings.watchEnabled &&
                !runtime.statePollTimer
            ) {
                scheduleAdaptiveStateCheck();
            }
        });

        scanAnomalyDom();
    }

    function stopWatcher() {
        settings.watchEnabled = false;
        runtime.manualStatus = "Наблюдение остановлено";

        if (runtime.statePollTimer) {
            clearTimeout(runtime.statePollTimer);
            runtime.statePollTimer = null;
        }

        runtime.nextStateCheckAt = 0;
        runtime.statePollMode = "off";

        restoreSiteVisibility();
        stopBackgroundKeepAlive();
        clearHighlightFromOldButtons();

        saveSettings();
    }

    function toggleSound() {
        settings.soundEnabled = !settings.soundEnabled;

        if (settings.soundEnabled) {
            runtime.manualStatus = "Звук включён";
            primeAudio();
        } else {
            runtime.manualStatus = "Звук отключён";

            if (runtime.audio) {
                try {
                    runtime.audio.pause();
                    runtime.audio.currentTime = 0;
                } catch {}
            }
        }

        saveSettings();
    }

    function formatCooldown() {
        const ms = Math.max(
            0,
            runtime.cooldownUntil - Date.now()
        );

        if (ms <= 0) return "0";

        const sec = Math.ceil(ms / 1000);

        if (sec < 60) return `${sec}с`;

        const min = Math.floor(sec / 60);
        const rest = sec % 60;

        return `${min}м ${rest}с`;
    }

    function formatNextStateCheck() {
        if (!settings.watchEnabled) return "off";
        if (!runtime.nextStateCheckAt) return "—";

        const ms = Math.max(
            0,
            runtime.nextStateCheckAt - Date.now()
        );

        const sec = Math.ceil(ms / 1000);

        if (sec <= 0) {
            return `сейчас · ${runtime.statePollMode}`;
        }

        if (sec < 60) {
            return `${sec}с · ${runtime.statePollMode}`;
        }

        const min = Math.floor(sec / 60);
        const rest = sec % 60;

        return `${min}м ${rest}с · ${runtime.statePollMode}`;
    }

    function formatAudioState() {
        if (!settings.soundEnabled) return "OFF";

        if (runtime.audioState === "ready") {
            return "ON · ready";
        }

        if (runtime.audioState === "blocked") {
            return "ON · браузер блокирует";
        }

        if (runtime.audioState === "resource-error") {
            return "ON · ошибка ресурса";
        }

        return "ON · ожидает жест";
    }

    function applyStoredPanelPosition() {
        if (!panel) return;

        const { x, y } = settings.panel;

        if (
            Number.isFinite(x) &&
            Number.isFinite(y)
        ) {
            panel.style.left = `${x}px`;
            panel.style.top = `${y}px`;
            panel.style.right = "auto";
            panel.style.bottom = "auto";
        } else {
            panel.style.right = "12px";
            panel.style.bottom = "12px";
            panel.style.left = "auto";
            panel.style.top = "auto";
        }
    }

    function clampPanelToViewport() {
        if (!panel) return;

        if (
            !Number.isFinite(settings.panel.x) ||
            !Number.isFinite(settings.panel.y)
        ) {
            return;
        }

        const rect = panel.getBoundingClientRect();

        settings.panel.x = clamp(
            settings.panel.x,
            0,
            Math.max(0, window.innerWidth - rect.width)
        );

        settings.panel.y = clamp(
            settings.panel.y,
            0,
            Math.max(0, window.innerHeight - rect.height)
        );

        applyStoredPanelPosition();
        saveSettings();
    }

    function installPanelDrag(header) {
        let dragging = false;
        let pointerId = null;
        let dx = 0;
        let dy = 0;

        header.addEventListener("pointerdown", event => {
            if (
                event.button !== 0 ||
                event.target.closest("button")
            ) {
                return;
            }

            dragging = true;
            pointerId = event.pointerId;

            const rect = panel.getBoundingClientRect();

            dx = event.clientX - rect.left;
            dy = event.clientY - rect.top;

            settings.panel.x = rect.left;
            settings.panel.y = rect.top;

            panel.style.right = "auto";
            panel.style.bottom = "auto";

            try {
                header.setPointerCapture(pointerId);
            } catch {}

            event.preventDefault();
        });

        header.addEventListener("pointermove", event => {
            if (!dragging || event.pointerId !== pointerId) {
                return;
            }

            const rect = panel.getBoundingClientRect();

            const x = clamp(
                event.clientX - dx,
                0,
                Math.max(0, window.innerWidth - rect.width)
            );

            const y = clamp(
                event.clientY - dy,
                0,
                Math.max(0, window.innerHeight - rect.height)
            );

            settings.panel.x = x;
            settings.panel.y = y;

            panel.style.left = `${x}px`;
            panel.style.top = `${y}px`;

            event.preventDefault();
        });

        const endDrag = event => {
            if (!dragging) return;

            if (
                event.pointerId != null &&
                pointerId != null &&
                event.pointerId !== pointerId
            ) {
                return;
            }

            dragging = false;

            try {
                header.releasePointerCapture(pointerId);
            } catch {}

            pointerId = null;
            saveSettings();
        };

        header.addEventListener("pointerup", endDrag);
        header.addEventListener("pointercancel", endDrag);
    }

    function ensurePanel() {
        if (panelRecreateInProgress) return;
        if (!document.body) return;

        const connected =
            panel?.isConnected &&
            document.getElementById(
                "aon-anomaly-watcher-v12-panel"
            ) === panel;

        if (connected) return;

        panelRecreateInProgress = true;
        clearPanelRefs();

        try {
            createPanel();
        } finally {
            panelRecreateInProgress = false;
        }
    }

    function startPanelWatchdog() {
        if (panelWatchInterval) {
            clearInterval(panelWatchInterval);
        }

        panelWatchInterval = setInterval(
            ensurePanel,
            PANEL_WATCH_MS
        );
    }

    // ------------------------------------------------------------
    // Visibility events / startup / cleanup
    // ------------------------------------------------------------

    document.addEventListener("visibilitychange", () => {
        if (!settings.watchEnabled) return;

        if (!isActuallyHidden()) {
            fetchState("visibility").finally(() => {
                if (
                    settings.watchEnabled &&
                    !runtime.statePollTimer
                ) {
                    scheduleAdaptiveStateCheck();
                }
            });
        }
    });

    function startup() {
        if (settings.watchEnabled) {
            startBackgroundKeepAlive();

            setTimeout(() => {
                fetchState("startup").finally(() => {
                    if (
                        settings.watchEnabled &&
                        !runtime.statePollTimer
                    ) {
                        scheduleAdaptiveStateCheck();
                    }
                });
            }, 900);
        }
    }

    if (document.readyState === "loading") {
        document.addEventListener(
            "DOMContentLoaded",
            () => {
                createPanel();
                startPanelWatchdog();
                startup();
            },
            { once: true }
        );
    } else {
        createPanel();
        startPanelWatchdog();
        startup();
    }

    window.addEventListener("pagehide", () => {
        if (panelWatchInterval) {
            clearInterval(panelWatchInterval);
            panelWatchInterval = null;
        }

        if (runtime.statePollTimer) {
            clearTimeout(runtime.statePollTimer);
            runtime.statePollTimer = null;
        }

        restoreSiteVisibility();
        stopBackgroundKeepAlive();
    });
