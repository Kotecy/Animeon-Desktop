    let trackedNick = '';

    let requestInFlight = false;

        }

        #aoxp-banner-main {
            font-size: clamp(38px, 6vw, 72px);

    const panel = document.createElement('div');
    panel.id = 'aoxp-panel';
    panel.innerHTML = `
        <div id="aoxp-header">
            <div id="aoxp-title">XP Monitor</div>
            <button id="aoxp-minimize" type="button" title="Свернуть">−</button>
            <div id="aoxp-compact-progress-wrap"><div id="aoxp-compact-progress"></div></div>
        </div>
        <div id="aoxp-body">
            <input id="aoxp-input" type="text" autocomplete="off" spellcheck="false" placeholder="Ник пользователя → Enter">
            <div id="aoxp-status">
                <span id="aoxp-dot"></span>
                <span id="aoxp-status-text">Введите ник и нажмите Enter</span>
            </div>
            <div id="aoxp-grid">
                <div class="aoxp-card">
                    <div class="aoxp-label">Текущий XP</div>
                    <div class="aoxp-value" id="aoxp-current">—</div>
                </div>
                <div class="aoxp-card">
                    <div class="aoxp-label">Получено</div>
                    <div class="aoxp-value" id="aoxp-gained">+0</div>
                </div>
                <div class="aoxp-card">
                    <div class="aoxp-label">Следующая цель</div>
                    <div class="aoxp-value" id="aoxp-target">—</div>
                </div>
                <div class="aoxp-card">
                    <div class="aoxp-label">Осталось</div>
                    <div class="aoxp-value" id="aoxp-remaining">—</div>
                </div>
            </div>
            <div id="aoxp-progress-scale">
                <span id="aoxp-progress-start">—</span>
                <span id="aoxp-progress-target">—</span>
            </div>
            <div id="aoxp-progress-wrap"><div id="aoxp-progress"></div></div>
            <div id="aoxp-delta"></div>
        </div>
    `;
    document.body.appendChild(panel);

    const banner = document.createElement('div');
    banner.id = 'aoxp-banner';
    banner.innerHTML = `
        <div id="aoxp-banner-title"></div>
        <div id="aoxp-banner-main"></div>
        <div id="aoxp-banner-sub"></div>
    `;
    document.body.appendChild(banner);

    const header = panel.querySelector('#aoxp-header');
    const minimize = panel.querySelector('#aoxp-minimize');
    const title = panel.querySelector('#aoxp-title');
    const input = panel.querySelector('#aoxp-input');
    const status = panel.querySelector('#aoxp-status');
    const statusText = panel.querySelector('#aoxp-status-text');
    const currentEl = panel.querySelector('#aoxp-current');
    const gainedEl = panel.querySelector('#aoxp-gained');
    const targetEl = panel.querySelector('#aoxp-target');
    const remainingEl = panel.querySelector('#aoxp-remaining');
    const progressEl = panel.querySelector('#aoxp-progress');
    const compactProgressEl = panel.querySelector('#aoxp-compact-progress');
    const progressStartEl = panel.querySelector('#aoxp-progress-start');
    const progressTargetEl = panel.querySelector('#aoxp-progress-target');
    const deltaEl = panel.querySelector('#aoxp-delta');
    const bannerTitle = banner.querySelector('#aoxp-banner-title');
    const bannerMain = banner.querySelector('#aoxp-banner-main');
    const bannerSub = banner.querySelector('#aoxp-banner-sub');

    function setStatus(kind, text) {
        status.className = kind ? `aoxp-${kind}` : '';
        statusText.textContent = text;
    }

    function nextRoundAbove(xp) {
        return (Math.floor(xp / ROUND_STEP) + 1) * ROUND_STEP;
    }

    function hideBanner() {
        banner.classList.remove('aoxp-show', 'aoxp-reached');
    }

    function showNearBanner(remaining, target) {
        if (reachedTimer) {
            clearTimeout(reachedTimer);
            reachedTimer = null;
        }
        banner.classList.remove('aoxp-reached');
        bannerTitle.textContent = '⚠ Скоро круглый XP';
        bannerMain.textContent = `${fmt(remaining)} XP`;
        bannerSub.textContent = `до ${fmt(target)}`;
        banner.classList.add('aoxp-show');
    }

    function showReachedBanner(reachedTarget, newTarget) {
        if (reachedTimer) clearTimeout(reachedTimer);
        banner.classList.add('aoxp-reached', 'aoxp-show');
        bannerTitle.textContent = '🎯 Круглый XP достигнут';
        bannerMain.textContent = fmt(reachedTarget);
        bannerSub.textContent = `Теперь считаем до ${fmt(newTarget)}`;
        reachedTimer = setTimeout(() => {
            reachedTimer = null;
            updateWarning();
        }, 2800);
    }

    function updateWarning() {
        if (reachedTimer) return;

        if (!Number.isFinite(lastXp) || !Number.isFinite(targetXp)) {
            hideBanner();
            return;
        }

        const remaining = targetXp - lastXp;
        if (remaining > 0 && remaining <= WARNING_AT) {
            showNearBanner(remaining, targetXp);
        } else {
            hideBanner();
        }
    }

    function updateUi(delta = null) {
        if (!Number.isFinite(lastXp)) return;

        const gained = Number.isFinite(startXp) ? lastXp - startXp : 0;
        const remaining = targetXp - lastXp;
        const roundStart = targetXp - ROUND_STEP;
        const withinThousand = Math.max(0, lastXp - roundStart);
        const progress = Math.max(0, Math.min(100, (withinThousand / ROUND_STEP) * 100));

        currentEl.textContent = fmt(lastXp);
        gainedEl.textContent = `${gained >= 0 ? '+' : ''}${fmt(gained)}`;
        targetEl.textContent = fmt(targetXp);
        remainingEl.textContent = fmt(Math.max(0, remaining));
        remainingEl.classList.toggle('aoxp-near', remaining > 0 && remaining <= WARNING_AT);
        progressStartEl.textContent = fmt(roundStart);
        progressTargetEl.textContent = fmt(targetXp);
        progressEl.style.width = `${progress}%`;
        compactProgressEl.style.width = `${progress}%`;

        if (delta !== null && delta !== 0) {
            deltaEl.textContent = `Последнее изменение: ${delta > 0 ? '+' : ''}${fmt(delta)} XP`;
        }

        title.textContent = panel.classList.contains('aoxp-collapsed')
            ? `${trackedNick}: ${fmt(lastXp)} · до ${fmt(Math.max(0, remaining))}`
            : `XP Monitor · ${trackedNick}`;
    }

    function applyXp(xp) {
        if (!Number.isFinite(xp)) return;

        const previous = lastXp;

        if (startXp === null) {
            startXp = xp;
            lastXp = xp;
            targetXp = nextRoundAbove(xp);
            updateUi();
            updateWarning();
            return;
        }

        if (xp === lastXp) {
            updateWarning();
            return;
        }

        lastXp = xp;

        if (!Number.isFinite(targetXp)) {
            targetXp = nextRoundAbove(xp);
        }

        if (xp >= targetXp) {
            const reachedTarget = targetXp;
            targetXp = nextRoundAbove(xp);
            updateUi(previous === null ? null : xp - previous);
            showReachedBanner(reachedTarget, targetXp);
            return;
        }

        updateUi(previous === null ? null : xp - previous);
        updateWarning();
    }

    function extractXp(data) {
        const user = data?.user && typeof data.user === 'object' ? data.user : data;
        const xp = Number(user?.total_xp);
        return Number.isFinite(xp) ? xp : null;
    }

    async function readProfile(generation = requestGeneration) {
        if (!trackedNick || requestInFlight) return;

        requestInFlight = true;
        try {
            const response = await fetch(
                `/api/users/${encodeURIComponent(trackedNick)}?xp_monitor=${Date.now()}`,
                {
                    credentials: 'include',
                    cache: 'no-store',
                    headers: { Accept: 'application/json' }
                }
            );

            if (generation !== requestGeneration) return;

            if (!response.ok) {
                setStatus('error', response.status === 404 ? 'Пользователь не найден' : `Ошибка API: ${response.status}`);
                return;
            }

            const data = await response.json();
            if (generation !== requestGeneration) return;

            const xp = extractXp(data);
            if (xp === null) {
                setStatus('error', 'В ответе API нет total_xp');
                return;
            }

            setStatus('ok', `Отслеживается: ${trackedNick}`);
            applyXp(xp);
        } catch {
            if (generation === requestGeneration) {
                setStatus('error', 'Нет ответа от сайта');
            }
        } finally {
            requestInFlight = false;
        }
    }

    function resetForNick(nick) {
        requestGeneration += 1;
        trackedNick = nick;
        startXp = null;
        lastXp = null;
        targetXp = null;
        currentEl.textContent = '—';
        gainedEl.textContent = '+0';
        targetEl.textContent = '—';
        remainingEl.textContent = '—';
        remainingEl.classList.remove('aoxp-near');
        progressEl.style.width = '0%';
        compactProgressEl.style.width = '0%';
        progressStartEl.textContent = '—';
        progressTargetEl.textContent = '—';
        deltaEl.textContent = '';
        title.textContent = `XP Monitor · ${trackedNick}`;
        hideBanner();
        setStatus('wait', `Получаю XP: ${trackedNick}…`);
        readProfile(requestGeneration);
    }

    input.addEventListener('keydown', (event) => {
        if (event.key !== 'Enter') return;
        const nick = input.value.trim().replace(/^@/, '');
        if (!nick) {
            setStatus('error', 'Введите ник пользователя');
            return;
        }
        input.blur();
        resetForNick(nick);
    });

    minimize.addEventListener('mousedown', (event) => event.stopPropagation());
    minimize.addEventListener('click', (event) => {
        event.stopPropagation();
        panel.classList.toggle('aoxp-collapsed');
        const collapsed = panel.classList.contains('aoxp-collapsed');
        minimize.textContent = collapsed ? '+' : '−';
        minimize.title = collapsed ? 'Развернуть' : 'Свернуть';
        if (Number.isFinite(lastXp)) updateUi();
        else title.textContent = collapsed ? 'XP Monitor' : 'XP Monitor';
    });

    let dragging = false;
    let dragOffsetX = 0;
    let dragOffsetY = 0;

    header.addEventListener('mousedown', (event) => {
        if (event.button !== 0 || event.target === minimize) return;
        const rect = panel.getBoundingClientRect();
        dragging = true;
        dragOffsetX = event.clientX - rect.left;
        dragOffsetY = event.clientY - rect.top;
        event.preventDefault();
    });

    window.addEventListener('mousemove', (event) => {
        if (!dragging) return;

        const maxLeft = Math.max(0, window.innerWidth - panel.offsetWidth);
        const maxTop = Math.max(0, window.innerHeight - panel.offsetHeight);
        const left = Math.min(maxLeft, Math.max(0, event.clientX - dragOffsetX));
        const top = Math.min(maxTop, Math.max(0, event.clientY - dragOffsetY));

        panel.style.left = `${left}px`;
        panel.style.top = `${top}px`;
        panel.style.right = 'auto';
        panel.style.bottom = 'auto';
    });

    window.addEventListener('mouseup', () => {
        dragging = false;
    });

    window.addEventListener('resize', () => {
        const rect = panel.getBoundingClientRect();
        const maxLeft = Math.max(0, window.innerWidth - panel.offsetWidth);
        const maxTop = Math.max(0, window.innerHeight - panel.offsetHeight);
        panel.style.left = `${Math.min(maxLeft, Math.max(0, rect.left))}px`;
        panel.style.top = `${Math.min(maxTop, Math.max(0, rect.top))}px`;
    });

    setInterval(() => readProfile(requestGeneration), POLL_MS);
})();
